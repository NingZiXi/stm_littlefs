cmake_minimum_required(VERSION 3.22)
project(stm_littlefs_tests LANGUAGES C CXX)
enable_testing()
get_filename_component(COMPONENT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(FLASH_SOURCE_DIR "${COMPONENT}/../stm_flash" CACHE PATH "Actual stm_flash headers")
set(COMMON_SOURCE_DIR "${COMPONENT}/../stm_common" CACHE PATH "Actual stm_common headers")
add_library(stm_flash INTERFACE)
target_include_directories(stm_flash INTERFACE "${FLASH_SOURCE_DIR}" "${COMMON_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(stm_flash INTERFACE STM_FLASH_HAL_HEADER="fake_hal.h")
add_subdirectory("${COMPONENT}" component)
target_compile_options(stm_littlefs PRIVATE -Wall -Wextra -Werror)
# 分配失败注入只作用于适配层，验证部分分配的回收。
target_compile_definitions(stm_littlefs PRIVATE calloc=test_calloc free=test_free)
add_executable(adapter_tests test_littlefs.c headers.cpp)
target_compile_features(adapter_tests PRIVATE c_std_11 cxx_std_17)
target_compile_options(adapter_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(adapter_tests PRIVATE stm_littlefs)
add_test(NAME adapter_tests COMMAND adapter_tests)
