cmake_minimum_required(VERSION 3.22)
project(stm_littlefs VERSION 1.0.0 LANGUAGES C)
include(FetchContent)

if(NOT TARGET stm_flash)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_flash/CMakeLists.txt")
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../stm_flash" "${CMAKE_CURRENT_BINARY_DIR}/stm_flash")
    else()
        option(STM_LITTLEFS_FETCH_FLASH "Fetch stm_flash if not provided locally" ON)
        set(STM_LITTLEFS_FLASH_GIT_REPOSITORY "https://github.com/NingZiXi/stm_flash.git" CACHE STRING "stm_flash repository")
        if(NOT STM_LITTLEFS_FETCH_FLASH)
            message(FATAL_ERROR "Provide stm_flash target/sibling or enable STM_LITTLEFS_FETCH_FLASH")
        endif()
        FetchContent_Declare(stm_flash
            GIT_REPOSITORY "${STM_LITTLEFS_FLASH_GIT_REPOSITORY}"
            GIT_TAG v3.0.0)
        FetchContent_MakeAvailable(stm_flash)
    endif()
endif()
if(NOT TARGET stm_flash)
    message(FATAL_ERROR "stm_flash dependency did not provide its target")
endif()

# 官方仓库不依赖 STM32；源码仅下载到构建目录，不复制进组件。
if(NOT TARGET littlefs)
    set(STM_LITTLEFS_SOURCE_DIR "" CACHE PATH "Offline official LittleFS source directory")
    option(STM_LITTLEFS_FETCH "Fetch official LittleFS when no target/source is supplied" ON)
    set(STM_LITTLEFS_GIT_REPOSITORY "https://github.com/littlefs-project/littlefs.git" CACHE STRING "Official LittleFS repository or trusted mirror")
    if(STM_LITTLEFS_SOURCE_DIR)
        set(lfs_source "${STM_LITTLEFS_SOURCE_DIR}")
    else()
        if(NOT STM_LITTLEFS_FETCH)
            message(FATAL_ERROR "Provide littlefs target/STM_LITTLEFS_SOURCE_DIR or enable STM_LITTLEFS_FETCH")
        endif()
        FetchContent_Declare(littlefs
            GIT_REPOSITORY "${STM_LITTLEFS_GIT_REPOSITORY}"
            GIT_TAG adad0fbbcf5382c20978d07f94f9c13be9041c1b # v2.11.2
            SOURCE_SUBDIR stm_littlefs_no_upstream_cmake)
        FetchContent_MakeAvailable(littlefs)
        set(lfs_source "${littlefs_SOURCE_DIR}")
    endif()
    foreach(source lfs.c lfs.h lfs_util.c lfs_util.h)
        if(NOT EXISTS "${lfs_source}/${source}")
            message(FATAL_ERROR "Missing LittleFS source: ${lfs_source}/${source}")
        endif()
    endforeach()
    add_library(littlefs STATIC "${lfs_source}/lfs.c" "${lfs_source}/lfs_util.c")
    target_include_directories(littlefs PUBLIC "${lfs_source}")
    target_compile_features(littlefs PUBLIC c_std_11)
    target_compile_definitions(littlefs PUBLIC LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
endif()

add_library(stm_littlefs STATIC stm_littlefs.c)
target_include_directories(stm_littlefs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(stm_littlefs PUBLIC c_std_11)
target_link_libraries(stm_littlefs PUBLIC stm_flash littlefs)
